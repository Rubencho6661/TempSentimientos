Creación de la variable target: sentiment_class

# Eliminar filas nulas en columnas críticas
df = df.dropna(subset=['content', 'sentiment_polarity', 'createdAt'])

# Crear columna categórica
def categorize_sentiment(polarity):
    if polarity > 0:
        return 'positivo'
    elif polarity < 0:
        return 'negativo'
    else:
        return 'neutro'

df['sentiment_class'] = df['sentiment_polarity'].apply(categorize_sentiment)

Preprocesamiento de texto

import re
from sklearn.feature_extraction.text import TfidfVectorizer
import nltk
nltk.download('stopwords')
from nltk.corpus import stopwords

stop_words = stopwords.words('english')

def clean_text(text):
    text = text.lower()
    text = re.sub(r"http\S+|www\S+|https\S+", '', text)
    text = re.sub(r'\W', ' ', text)
    text = re.sub(r'\s+', ' ', text)
    return text

df['content_clean'] = df['content'].apply(clean_text)

# Vectorizador
text_features = 'content_clean'
text_transformer = TfidfVectorizer(stop_words=stop_words, max_features=5000)

Análisis de Tendencias a lo Largo del Tiempo

# Crear columna con fecha sin hora
df['date'] = df['createdAt'].dt.date

# Contar cantidad de tweets por sentimiento por día
trend = df.groupby(['date', 'sentiment_class']).size().unstack(fill_value=0)

# Normalizar por total de tweets diarios (opcional)
trend_norm = trend.div(trend.sum(axis=1), axis=0)
