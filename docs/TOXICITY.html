1. Limpieza del texto y metadatos

Code 
lemmatizer = WordNetLemmatizer()
stop_words = set(stopwords.words('spanish'))  # usar idioma según contenido

def limpiar_texto(text):
    text = str(text).lower()
    text = re.sub(r"http\S+|www\S+|https\S+", '', text)       # eliminar URLs
    text = re.sub(r'@\w+|#\w+', '', text)                     # eliminar menciones y hashtags
    text = text.translate(str.maketrans('', '', string.punctuation))  # eliminar puntuación
    tokens = nltk.word_tokenize(text)
    tokens = [lemmatizer.lemmatize(word) for word in tokens if word not in stop_words and word.isalpha()]
    return ' '.join(tokens)

df['texto_limpio'] = df['content'].apply(limpiar_texto)

2. Definir transformadores TF-IDF por tipo de variable

preparacion de columnas de texto para el ML

Code 
text_transformer = Pipeline(steps=[
    ('tfidf', TfidfVectorizer(max_features=2000, ngram_range=(1,2))),
    # Reduce dimensionality of TF-IDF to obtain a dense, lower-dimensional representation
    ('svd', TruncatedSVD(n_components=100, random_state=42))
])

# Categorización de variables OneHotEncoder
cat_transformer = Pipeline(steps=[
    # Return dense output so downstream resamplers (SMOTE) can operate
    ('onehot', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
])

# Escalamiento de variables numéricas
num_transformer = Pipeline(steps=[
    ('scaler', StandardScaler())
])

Combinar transformadores para la nueva database en un ColumnTransformer para modelacións

Code 
preprocessor = ColumnTransformer(transformers=[
    ('text', text_transformer, text_col),
    ('cat', cat_transformer, cat_cols),
    ('num', num_transformer, num_cols)
])

# Crear pipeline completo (solo preprocesamiento)
preprocessing_pipeline = Pipeline(steps=[
    ('preprocessor', preprocessor)
])
